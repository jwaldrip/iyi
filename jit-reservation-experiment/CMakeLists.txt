cmake_minimum_required(VERSION 3.20)
project(jit_reservation_probe LANGUAGES C CXX)
find_package(LLVM 20.1.7 EXACT REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} at ${LLVM_DIR}")
add_executable(jit_reservation_probe probe.cpp)
target_include_directories(jit_reservation_probe SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(jit_reservation_probe PRIVATE ${LLVM_DEFINITIONS})
target_compile_features(jit_reservation_probe PRIVATE cxx_std_17)
llvm_map_components_to_libnames(llvm_libs core executionengine mcjit native)
target_link_libraries(jit_reservation_probe PRIVATE ${llvm_libs})
